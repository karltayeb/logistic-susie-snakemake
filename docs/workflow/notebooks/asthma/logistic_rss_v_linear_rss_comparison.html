Logistic+RSS and Linear+RSS give similar results.

We fine-map 15 loci associated with asthma. We use three asthma phenotypes, childhood onset asthma (COA), adult onset asthma (AOA), and all asthma (All). We perform finemapping in two ways, using either summary statistics from logistic regression or linear regression. The same, SNPs, covariates, etc are included in both association tests. For the linear association tests we standardize the binary phenotype. For each phenotype we compute the in-sample LD matrix (the included individuals differ across the three phenotypes).

Ultimately we find that the the \(z\)-scores are remarkably similar between the two approaches. Consequently downstream fine-mapping approaches such as SuSiE-RSS, which take the in-sample LD and marginal association \(z\)-scores as input, give qualitatively similar results. For example, we find strong agreement in the PIPs reported in each approach.

Notes on PLINK2

We perform the association analysis and compute the LD matrices using PLINK2. In order to get the in-sample correlation matrix among genotypes we use the --r-unphased square ref-based flag when computing the LD matrix and --glm ... omit-ref option when computing summary statistics. This ensures that the LD and summary statistics are both computed encoding genotype as dosage of the alternate allele(s). Note: I didn’t explicitly filter for bi-allelic variants, I am assuming that the provided genotype files were already processed appropriately.

library(dplyr)
library(ggplot2)

REGIONS <- purrr::map_chr(list.files('results/asthma/rss_linear/all/', pattern='*.linear_rss_fit.rds'), ~stringr::str_split(.x, '\\.')[[1]][1])

load_linear_rss_fit <- function(region, pheno){
    path <- glue::glue('results/asthma/rss_linear/{pheno}/{region}.{pheno}.linear_rss_fit.rds')
    if(file.exists(path)){
        return(readRDS(path))
    }
}
load_logistic_rss_fit <- function(region, pheno){
    path <- glue::glue('results/asthma/rss_logistic/{pheno}/{region}.{pheno}.logistic_rss_fit.rds')
    if(file.exists(path)){
        return(readRDS(path))
    }
}

load_pvar <- function(region){
    path <- glue::glue('resources/asthma/geno/{region}.pvar')
    readr::read_tsv(path)
}


load_linear_ss <- function(region, pheno){
    path <- glue::glue('results/asthma/plink/{pheno}/gwas_{region}.asthma_{pheno}.glm.linear')
    readr::read_tsv(path)
}

load_logistic_ss <- function(region, pheno){
    path <- glue::glue('results/asthma/plink/{pheno}/gwas_{region}.asthma_{pheno}.glm.logistic.hybrid')
    readr::read_tsv(path)
}


res <- tidyr::crossing(region = REGIONS, pheno = c('all', 'aoa', 'coa')) %>%
    dplyr::rowwise() %>%
    dplyr::mutate(
        snp_data = list(load_pvar(region)),
        linear_fit = list(load_linear_rss_fit(region, pheno)),
        linear_ss = list(load_linear_ss(region, pheno)),
        logistic_fit = list(load_logistic_rss_fit(region, pheno)),
        logistic_ss = list(load_logistic_ss(region, pheno)))

res <- res %>%
    mutate(linear_converged = linear_fit$converged, logistic_converged=logistic_fit$converged)

all(res$linear_converged)
all(res$logistic_converged)
pip_data <- res %>%
    rowwise() %>%
    dplyr::mutate(
        linear_pip = list(linear_fit$pip),
        logistic_pip = list(logistic_fit$pip),
        converged = linear_fit$converged & logistic_fit$converged
    ) %>%
    ungroup() %>%
    # dplyr::filter(converged) %>%
    dplyr::select(region, pheno, snp_data, linear_pip, logistic_pip) %>%
    tidyr::unnest_wider(snp_data) %>%
    dplyr::rename(CHROM = `#CHROM`) %>%
    tidyr::unnest_longer(c(region, CHROM, POS, ID, REF, ALT, linear_pip, logistic_pip))

# Plot PIPS from logistic+rss against PIPs from linear+rss
pip_data %>%
    filter(linear_pip > 0.0001 | logistic_pip > 0.001) %>%
    ggplot(aes(x = linear_pip, y=logistic_pip)) + geom_point() + theme_bw() + geom_abline(intercept=0, slope=1, color='red')
# plot z-scores from logistic GWAS and z-score from linear GWAS
res %>%
    rowwise() %>%
    dplyr::mutate(
        linear_z= list(linear_ss$T_STAT),
        logistic_z = list(logistic_ss$Z_STAT)
    ) %>%
    ungroup() %>%
    dplyr::select(region, pheno, snp_data, linear_z, logistic_z) %>%
    tidyr::unnest_longer(c(linear_z, logistic_z)) %>%
    ggplot(aes(x=linear_z, y = logistic_z)) + geom_point() + geom_abline(intercept=0, slope=1, color='red') + theme_bw()
pip_data %>% dplyr::filter(abs(logistic_pip - linear_pip) > 0.1) %>%
    select(pheno, region, ID, CHROM, linear_pip, logistic_pip) %>%
    knitr::kable()
cs_data <- res %>%
    dplyr::rowwise() %>%
    dplyr::mutate(cs = list(linear_fit$sets)) %>%
    tidyr::unnest_wider(cs) %>%
    dplyr::ungroup() %>%
    dplyr::filter(!purrr::map_lgl(cs, is.null)) %>%
    tidyr::unnest_longer(c(cs, purity, cs_index)) %>%
    dplyr::rowwise() %>%
    dplyr::mutate(snp_data = list(snp_data[cs,]), pip = list(linear_fit$pip[cs])) %>%
    dplyr::select(-c(linear_fit, logistic_fit)) %>%
    tidyr::unnest_longer(c(snp_data, pip, cs))
plot_side_by_side <- function(region, pheno, y='PIP', ...){
    par(mfrow = c(1, 2))
    i <- which(res$region == region & res$pheno == pheno)[1]
    susieR::susie_plot(res$linear_fit[[i]], y=y, main='Linear', ...)
    susieR::susie_plot(res$logistic_fit[[i]], y=y, main='Logistic', ...)
}
